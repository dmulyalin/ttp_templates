config system interface
    edit "port1"
        set vdom "root"
        set ip 192.168.122.160 255.255.255.0
        set allowaccess ping https ssh
        set type physical
        set snmp-index 1
    next
    edit "naf.root"
        set vdom "root"
        set type tunnel
        set src-check disable
        set snmp-index 2
    next
    edit "l2t.root"
        set vdom "root"
        set type tunnel
        set snmp-index 3
    next
    edit "ssl.root"
        set vdom "root"
        set type tunnel
        set alias "SSL VPN interface"
        set snmp-index 4
    next
    edit "fortilink"
        set vdom "root"
        set fortilink enable
        set ip 10.255.1.1 255.255.255.0
        set allowaccess ping fabric
        set type aggregate
        set lldp-reception enable
        set lldp-transmission enable
        set snmp-index 5
    next
    edit "ambit-lo977"
        set vdom "root"
        set vrf 7
        set ip 198.51.100.77 255.255.255.255
        set allowaccess ping
        set type loopback
        set snmp-index 6
        config ipv6
            set ip6-address 2001:db8:77::77/128
            set ip6-allowaccess ping
        end
    next
    edit "ambit-v977"
        set vdom "root"
        set vrf 7
        set ip 192.0.2.160 255.255.255.0
        set allowaccess ping
        set device-identification enable
        set role lan
        set snmp-index 7
        config ipv6
            set ip6-address 2001:db8:977::160/64
            set ip6-allowaccess ping
        end
        set interface "port1"
        set vlanid 977
    next
